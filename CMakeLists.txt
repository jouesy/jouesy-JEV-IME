cmake_minimum_required(VERSION 3.20)
project(JEVIME VERSION 0.1.0 LANGUAGES CXX C)

option(JEV_PORTABLE_ONLY "Build the offline core and ranking checks without Windows TSF" OFF)
option(JEV_BUILD_TESTS "Build JEV and upstream Windows tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(WINMCBOPOMOFO_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated")
include(cmake/JevIdentity.cmake)
execute_process(
    COMMAND ${CMAKE_COMMAND}
        -DINPUT_CMAKE_FILE=${CMAKE_SOURCE_DIR}/CMakeLists.txt
        -DOUTPUT_RC_FILE=${WINMCBOPOMOFO_GENERATED_DIR}/WinMcBopomofoVersion.rcinc
        -P ${CMAKE_SOURCE_DIR}/cmake/GenerateVersionRc.cmake
    COMMAND_ERROR_IS_FATAL ANY
)
if(MSVC)
    add_compile_options(/W4 /WX- /EHsc /utf-8 /FS /GR)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(src/JevEngine)
if(JEV_BUILD_TESTS)
    enable_testing()
    add_executable(JevRankingTest tests/JevRankingTest.cpp)
    target_link_libraries(JevRankingTest PRIVATE JevRanking)
    add_test(NAME JevRankingTest COMMAND JevRankingTest)
    find_program(JEV_POWERSHELL NAMES pwsh powershell)
    if(JEV_POWERSHELL)
        add_test(NAME JevPackageTest
            COMMAND ${JEV_POWERSHELL} -NoProfile -File ${CMAKE_SOURCE_DIR}/tests/JevPackageTest.ps1)
    endif()
endif()

if(JEV_PORTABLE_ONLY)
    include(cmake/PortableEngine.cmake)
else()
    if(NOT WIN32)
        message(FATAL_ERROR "TSF requires Windows. For core checks use -DJEV_PORTABLE_ONLY=ON")
    endif()
    if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/OpenCC/CMakeLists.txt")
        message(FATAL_ERROR "Missing OpenCC. Run git submodule update --init --recursive")
    endif()
    set(BUILD_TESTING OFF CACHE BOOL "Disable third-party tests" FORCE)
    set(ENABLE_GETTEXT OFF CACHE BOOL "Disable Gettext" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static OpenCC" FORCE)
    set(USE_SYSTEM_DARTS OFF CACHE BOOL "Bundled darts" FORCE)
    set(USE_SYSTEM_MARISA OFF CACHE BOOL "Bundled marisa" FORCE)
    set(USE_SYSTEM_TINYXML2 OFF CACHE BOOL "Bundled tinyxml2" FORCE)
    set(BUILD_DOCUMENTATION OFF CACHE BOOL "No OpenCC docs" FORCE)
    # OpenCC's ENABLE_GTEST also registers third-party tests, even though its
    # EXCLUDE_FROM_ALL targets are not built by the JEV build. Load GTest
    # directly for JEV's regression tests without registering unbuilt tests.
    set(ENABLE_GTEST OFF CACHE BOOL "Disable OpenCC's own test suite" FORCE)
    if(JEV_BUILD_TESTS)
        set(BUILD_GMOCK OFF CACHE BOOL "JEV uses GTest only" FORCE)
        set(INSTALL_GTEST OFF CACHE BOOL "Do not install build-time test libraries" FORCE)
        add_subdirectory(third_party/OpenCC/deps/googletest-1.15.0
            third_party/googletest EXCLUDE_FROM_ALL)
    endif()
    option(SKIP_OPENCC_DICT "Skip OpenCC dictionaries for client-only builds" OFF)
    add_subdirectory(third_party/OpenCC EXCLUDE_FROM_ALL)
    add_subdirectory(src)
    if(JEV_BUILD_TESTS)
        include(tests/windows/CMakeLists.txt)
    endif()
endif()
